\section{Introduction}\label{introduction}

\begin{frame}{Research Question \& Motivation}
\phantomsection\label{research-question-motivation}
\begin{block}{Arnott, Kalesnik \& Linnainmaa (2023, RFS)}
\textbf{Factor Momentum}: factors that performed well recently tend to outperform in the future.\\
\medskip
Key claims: SR $\approx$ 0.5--0.9 (1963--2021), robust to factor universe, highly diversifying.
\end{block}

\bigskip

\textbf{Our replication asks three questions:}

\begin{enumerate}
\item[\textbf{P1}] Can we replicate factor-level momentum using daily JKP returns?
\item[\textbf{P2}] What does the strategy look like at the single-stock level?
\item[\textbf{P2}] Does it survive real-world frictions (TC, borrow costs, dividend tax)?
\end{enumerate}
\end{frame}

\section{Part 2: Single-Stock Level}\label{part-2-single-stock-level}

\begin{frame}{Phase 2 Overview}
\phantomsection\label{phase-2-overview}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{alertblock}{1) Factor Replication via JKP}
\begin{itemize}
\item 39 JKP direction-corrected factors (USA)
\item Stock-level long/short weights saved at each monthly rebalance
\item 29,586 unique stocks (PERMNOs), 1926--2024
\end{itemize}
\end{alertblock}
\begin{alertblock}{3) Net Position Netting}
\begin{itemize}
\item Stock may be long in one factor, short in another
\item Net weight = sum of signed contributions across factors
\item Critical to apply frictions on \textbf{net} exposure
\end{itemize}
\end{alertblock}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{2) Factor Momentum Signal (CS\_LO\_25)}
\begin{itemize}
\item Monthly calendar rebalancing (not daily)
\item Prior 1/6/12 calendar month return as signal
\item Long top 25\% of factors by momentum
\item 1-day implementation lag at each rebalance
\end{itemize}
\end{alertblock}
\begin{alertblock}{4) Frictions Applied}
\begin{itemize}
\item Transaction Costs (power-law in ME)
\item Short-sale Borrow Costs
\item Dividend Withholding Tax (Austrian 27.5\%)
\end{itemize}
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Phase 2 Data \& Engineering}
\phantomsection\label{phase-2-data-engineering}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Data Pipeline}
\begin{itemize}
\item \texttt{usa\_factor\_weights.parquet}: 87M rows (stock $\times$ factor $\times$ month)
\item \texttt{USA\_stocks\_char.parquet}: returns, ME, dividends
\item \texttt{USA\_daily\_rets.parquet}: daily factor returns for lag adjustment
\item Key fix: EOM date alignment (calendar vs.\ last trading day)
\end{itemize}
\end{block}
\begin{block}{Phase 2 Master File}
\begin{itemize}
\item 104M rows: strategy $\times$ lookback $\times$ stock $\times$ month
\item 30 combinations: 5 lookbacks $\times$ 6 strategies
\item 743 months: Jul 1963 -- Nov 2024
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Critical Bug Fixed}
\textbf{EOM date mismatch}: monthly factor returns used last \emph{trading} day as end-of-month, but stock weights use calendar EOM. When month-end falls on a weekend ($\sim$225/744 months), inner join silently drops those months.\\
\medskip
$\Rightarrow$ Original run: \textbf{518 months} (1985--2024)\\
$\Rightarrow$ Fixed (\texttt{.dt.month\_end()}): \textbf{743 months} (1963--2024)
\end{block}
\end{column}
\end{columns}

\begin{frame}{1-Day Implementation Lag}
\phantomsection\label{day-implementation-lag}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{block}{Why a Lag Matters}
\begin{itemize}
\item Signal observed at end-of-month $t$
\item Earliest trade possible: Day 1 of month $t+1$
\item Day 1 often captures $\approx 60\%$ of month's mean-reversion
\item Cost of lag $\approx$ \textbf{1.6--1.9\% pa} in annual returns
\end{itemize}
\end{block}
\begin{block}{Implementation}
\begin{itemize}
\item Use factor-level lag-adjusted returns as proxy
\item Lag-adj factor return = full-month return $-$ day-1 return
\item Stock-level lag-adj return proxied from factor-level lag correction
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.42\textwidth}

\includegraphics[width=\linewidth]{ILab_Factor_Momentum_Beamer_files/figure-beamer/lag-waterfall-1} 
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Transaction Costs Model}
\phantomsection\label{transaction-costs-model}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{block}{Power-Law TC Function}
$$TC_i^{\text{bps}} = \min\!\left(100,\; 5 + 1 \cdot \left(\frac{\text{ME}_{\text{ref}}}{\max(ME_i, 1)}\right)^{0.25}\right)$$
where $\text{ME}_{\text{ref}}$ = median ME of top-500 stocks.
\end{block}
\begin{itemize}
\item Large-cap ($ME \approx \text{ME}_{\text{ref}}$): $\approx 6$ bps one-way
\item Small-cap ($ME = 0.01 \times \text{ME}_{\text{ref}}$): $\approx 8$ bps one-way
\item Micro-cap: capped at 100 bps
\item Applied as \textbf{round-trip} on each rebalanced position
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\begin{block}{SC: D'Avolio (2002)}
Short borrow costs (annualized):
$$SC_i = \min\!\left(\tfrac{500}{12},\; \tfrac{25 + 10 \cdot (ME_{\text{ref}}/ME_i)^{0.5}}{12}\right) \text{bps/mo}$$
D'Avolio: 91\% of loans $<$1\%/yr; VW average $\approx$ 17--24 bps/yr
\end{block}
\begin{block}{Dividend Tax (Austrian)}
27.5\% Kapitalertragsteuer on short leg manufactured dividends (non-deductible, \S20 EStG):
$$\text{Tax}_i = |w_i| \cdot \tfrac{\text{div12m\_me}_i}{12} \cdot 0.275$$
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Drift-Band Rebalancing}\label{drift-band-rebalancing}

\begin{frame}{Motivation: Reducing Turnover}
\phantomsection\label{motivation-reducing-turnover}
\begin{block}{Problem: High Monthly Turnover Drives TC Costs}
Average monthly turnover $\approx$ 36\% $\Rightarrow$ annual turnover $\approx$ 4.3$\times$ gross portfolio.\\
At 6 bps per trade for large-caps, this generates $\approx$ 0.55\% TC drag per year --- the single largest friction.
\end{block}
\bigskip
\begin{block}{Drift-Band Rebalancing (Threshold Rule)}
Instead of fully rebalancing every month, \textbf{only trade a stock if its target weight has changed by more than a threshold $\tau$}:
$$\text{Trade stock } i \text{ at month } t \iff |w_i^{\text{target}}(t) - w_i^{\text{held}}(t-1)| > \tau$$
\begin{itemize}
\item Small $\tau = 0$: always trade (baseline = monthly full rebalancing)
\item Larger $\tau$: skip small adjustments, save TC; accept mild signal degradation
\end{itemize}
\end{block}
\begin{alertblock}{Test thresholds: 0, 5, 10, 15, 20, 30, 50 bps}
\end{alertblock}
\end{frame}

\section{Lessons Learned}\label{lessons-learned}

\begin{frame}{Lessons Learned}
\phantomsection\label{lessons-learned-1}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Single-Stock Implementation}
Factor-level backtests overstate implementability; moving to single-name replication exposed concentration and idiosyncratic risk masked by factor aggregation.
\end{block}

\begin{block}{EOM Date Bug}
A subtle mismatch between trading-day EOM and calendar EOM silently dropped $\sim$225/744 months --- cutting the sample from 743 to 518 months. Always verify date alignment in multi-source joins.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Frictions Are Material}
Transaction costs alone absorb $\sim$0.55\% pa. Day-1 lag costs a further $\sim$1.6\% pa. Total drag from theory to net: $\sim$2.4\% pa on returns and $\sim$0.3 on Sharpe.
\end{block}
\begin{block}{Computational Scale}
104M rows in master file; required WU HPC cluster (SLURM), Python + Polars for speed, Arrow/Parquet for storage efficiency. R frictions analysis on filtered data ($\sim$3.5M rows per strategy) takes $\sim$10 min.
\end{block}
\begin{block}{Drift-Band: Signal Requires Frequent Updates}
Tested threshold rebalancing (0--50 bps). Result: full monthly rebalancing is optimal. Skipping even 5 bps trades reduces turnover by 41\% but cuts gross SR from 0.66 to 0.53. Each month's micro-adjustment carries meaningful aggregate signal --- the strategy cannot be ``thinned.''
\end{block}
\end{column}
\end{columns}
\end{frame}
